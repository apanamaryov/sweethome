# Управление отоплением — план реализации

> ⚠️ **Обновление 2026-08-01:** структура изменена — не отдельный сервис, а модуль
> `modules/heating` в репозитории `sweethome` (см. спеку объединения
> `docs/superpowers/specs/2026-08-01-sweethome-unification-design.md`). Пути и
> имена workspace'ов в тасках читать через эту призму; перед исполнением план
> актуализировать под фактический каркас.

> **Для агентов:** ОБЯЗАТЕЛЬНЫЙ САБ-СКИЛЛ: используй `superpowers:subagent-driven-development`
> (рекомендуется) или `superpowers:executing-plans`, чтобы выполнять план задача за
> задачей. Шаги размечены чекбоксами (`- [ ]`).

**Цель:** заменить облачное управление отоплением Tuya (3 датчика + реле котла +
реле насоса) на свою систему из **пяти** ESP32-нод, MQTT-брокера и Node-сервиса на
Raspberry Pi, с веб-интерфейсом и уведомлениями в Telegram.

**Архитектура:** три комнатных датчика, нода котла и нода насоса на ESPHome общаются
через Mosquitto на Pi. Решение «греть или нет» принимает Node-сервис
`heating-service`, но нода котла подписана на температуры напрямую и умеет работать
сама по лестнице L0–L3, а нода насоса слушает состояние котла и живёт по своей
лестнице P0/P1. Авторизации у сервиса нет — её обеспечивает Caddy через
`forward_auth` к уже работающему `inverter-monitor`.

**Стек:** TypeScript, Node 24, Express 4, `ws`, `mqtt` 5, `node:sqlite`, Next 15 +
React 19 (статический экспорт), uplot, jest + ts-jest + supertest, ESPHome,
Mosquitto, Caddy, systemd.

**Спецификация:** `SPEC.md` в корне репозитория. План реализует её целиком.

## Глобальные ограничения

- Node `>=24` (на Pi стоит `v24.18.0`, локально нужен тот же — см. `.nvmrc`).
- Монорепо на npm workspaces: `shared`, `server`, `web`. Имя scope — `@heating`.
- TypeScript `strict: true`, модуль `commonjs`, target `ES2020` (как в `inverter-monitor`).
- Все комментарии, сообщения об ошибках, тексты интерфейса и уведомлений — на русском.
- Сервис слушает **только** `127.0.0.1:3002`. Любой запрос не с loopback отклоняется.
- MQTT-префикс — `heating`. Контракт топиков зафиксирован в `SPEC.md` §5 и менять
  его в обход спеки нельзя: на нём завязана прошивка.
- Топик `heating/cmd/heat` публикуется **без retain**. Все `heating/cfg/*` — **с retain**.
- Пороги L3 (8 °C и 45 °C) зашиты в прошивку ноды котла и не выносятся в настройки.
- Нода котла публикует `heating/boiler/sensor/heat/state` **каждые 10 с**, а не только
  при изменении: на свежести этого топика держится лестница ноды насоса.
- Нода насоса при потере состояния котла **выключает** насос (уровень P1). Насос
  обслуживает только тёплый пол, крутить его вслепую бессмысленно.
- Коммиты — по-русски, в формате `feat: ...` / `fix: ...` / `test: ...` / `chore: ...`.
- Деплой на Pi: `pi@192.168.1.112`, каталог `/home/pi/heating`, вход по ключу
  (`alexey@piligrim2` уже в `authorized_keys`).
- **На прод ничего не деплоим без явной просьбы.** Pi — домашний стенд, деплой на
  неё разрешён.

- [ ] **Шаг 4: Написать `settings.ts`**

```ts
import { Settings } from "./types";

export const DEFAULT_SETTINGS: Settings = {
  mainSensor: "livingroom",
  setpoint: 21,
  hysteresis: 0.5,
  minStateSeconds: 300,
  staleAfterSeconds: 1800,
  pumpOverrunSeconds: 900,
  pumpAntiseizeSeconds: 604800,
  telegramEnabled: true,
};

interface NumRule {
  label: string;
  min: number;
  max: number;
  /** разрешить 0 как «выключено» вне диапазона */
  zeroMeansOff?: boolean;
}

const NUM_RULES: Record<string, NumRule> = {
  setpoint: { label: "уставка", min: 5, max: 30 },
  hysteresis: { label: "гистерезис", min: 0.1, max: 3 },
  minStateSeconds: { label: "минимальное время в состоянии", min: 60, max: 3600 },
  staleAfterSeconds: { label: "порог устаревания данных", min: 300, max: 7200 },
  pumpOverrunSeconds: { label: "выбег насоса", min: 0, max: 3600 },
  pumpAntiseizeSeconds: {
    label: "летний прогон насоса",
    min: 3600,
    max: 2592000,
    zeroMeansOff: true,
  },
};

/**
 * Мержит частичный патч в базовые настройки, проверяя каждое поле.
 * Бросает Error с человеческим текстом — он уходит прямо в ответ API.
 */
export function validateSettings(
  patch: unknown,
  rooms: string[],
  base: Settings = DEFAULT_SETTINGS
): Settings {
  if (patch === null || typeof patch !== "object" || Array.isArray(patch)) {
    throw new Error("Настройки должны быть объектом");
  }
  const out: Settings = { ...base };

  for (const [key, value] of Object.entries(patch as Record<string, unknown>)) {
    if (key === "mainSensor") {
      if (typeof value !== "string" || !rooms.includes(value)) {
        throw new Error(`Неизвестный главный датчик: ${String(value)}`);
      }
      out.mainSensor = value;
      continue;
    }
    if (key === "telegramEnabled") {
      if (typeof value !== "boolean") {
        throw new Error("Поле telegramEnabled должно быть true или false");
      }
      out.telegramEnabled = value;
      continue;
    }
    const rule = NUM_RULES[key];
    if (!rule) throw new Error(`Неизвестная настройка: ${key}`);
    if (typeof value !== "number" || !Number.isFinite(value)) {
      throw new Error(`Поле «${rule.label}» должно быть числом`);
    }
    const offOk = rule.zeroMeansOff === true && value === 0;
    if (!offOk && (value < rule.min || value > rule.max)) {
      throw new Error(
        `Поле «${rule.label}» вне допустимого диапазона ${rule.min}…${rule.max}`
      );
    }
    (out as unknown as Record<string, number>)[key] = value;
  }

  return out;
}
```

```ts
import { Settings } from "@heating/shared";

export interface ThermostatInput {
  /** unix ms */
  now: number;
  /** температура главного датчика, null — не приходила ни разу */
  mainTemp: number | null;
  /** unix ms последнего значения главного датчика */
  mainTempAt: number | null;
  /** фактическое состояние сухого контакта, как его сообщила нода котла */
  currentHeat: boolean;
  /** unix ms последней смены состояния; null — история пуста */
  lastChangeAt: number | null;
}

export type Decision =
  | { command: "on" | "off"; reason: string }
  | { command: "none"; reason: string };

/**
 * Чистое решение термостата. Никаких сайд-эффектов и никакого времени изнутри —
 * `now` приходит снаружи, чтобы тесты не зависели от часов.
 *
 * `command: "none"` означает «сервис не в состоянии решать» — вызывающий код обязан
 * НЕ публиковать команду, чтобы нода котла ушла на свою лестницу деградации.
 */
export function decide(input: ThermostatInput, cfg: Settings): Decision {
  const { now, mainTemp, mainTempAt, currentHeat, lastChangeAt } = input;

  if (mainTemp === null || mainTempAt === null) {
    return { command: "none", reason: "нет данных главного датчика" };
  }
  if (now - mainTempAt > cfg.staleAfterSeconds * 1000) {
    const age = Math.round((now - mainTempAt) / 60000);
    return { command: "none", reason: `данные главного датчика устарели на ${age} мин` };
  }

  const half = cfg.hysteresis / 2;
  let want: boolean;
  if (mainTemp <= cfg.setpoint - half) want = true;
  else if (mainTemp >= cfg.setpoint + half) want = false;
  else want = currentHeat;

  if (
    want !== currentHeat &&
    lastChangeAt !== null &&
    now - lastChangeAt < cfg.minStateSeconds * 1000
  ) {
    const left = Math.ceil((cfg.minStateSeconds * 1000 - (now - lastChangeAt)) / 1000);
    return {
      command: currentHeat ? "on" : "off",
      reason: `защита горелки: переключение через ${left} с`,
    };
  }

  return {
    command: want ? "on" : "off",
    reason: want
      ? `${mainTemp.toFixed(1)} °C ниже уставки ${cfg.setpoint} °C`
      : `${mainTemp.toFixed(1)} °C не ниже уставки ${cfg.setpoint} °C`,
  };
}
```

```ts
import { DatabaseSync } from "node:sqlite";
import { DEFAULT_SETTINGS, Settings, validateSettings } from "@heating/shared";

export interface SampleRow {
  ts: number;
  node: string;
  temp: number | null;
  humidity: number | null;
  rssi: number | null;
}

export interface BoilerEventRow {
  ts: number;
  heat: boolean;
  pump: boolean;
  /** уровень ноды котла: L0…L3 */
  level: string;
  /** уровень ноды насоса: P0/P1 */
  pumpLevel: string;
  reason: string;
}

export interface HourlyRow {
  hour: number;
  node: string;
  tempAvg: number;
  tempMin: number;
  tempMax: number;
  humidityAvg: number | null;
}

/** YYYY-MM-DD в локальном часовом поясе хоста. */
function localDay(ms: number): string {
  const d = new Date(ms);
  const p = (x: number) => String(x).padStart(2, "0");
  return `${d.getFullYear()}-${p(d.getMonth() + 1)}-${p(d.getDate())}`;
}

export class Store {
  private db: DatabaseSync;

  constructor(file: string) {
    this.db = new DatabaseSync(file);
    this.db.exec("PRAGMA journal_mode = WAL");
    this.migrate();
  }

  private migrate(): void {
    this.db.exec(`
      CREATE TABLE IF NOT EXISTS samples (
        ts INTEGER NOT NULL,
        node TEXT NOT NULL,
        temp REAL,
        humidity REAL,
        rssi INTEGER,
        PRIMARY KEY (ts, node)
      );
      CREATE TABLE IF NOT EXISTS samples_hourly (
        hour INTEGER NOT NULL,
        node TEXT NOT NULL,
        temp_avg REAL,
        temp_min REAL,
        temp_max REAL,
        humidity_avg REAL,
        PRIMARY KEY (hour, node)
      );
      CREATE TABLE IF NOT EXISTS boiler_events (
        ts INTEGER PRIMARY KEY,
        heat INTEGER NOT NULL,
        pump INTEGER NOT NULL,
        level TEXT NOT NULL,
        pump_level TEXT NOT NULL,
        reason TEXT NOT NULL
      );
      CREATE TABLE IF NOT EXISTS settings (
        key TEXT PRIMARY KEY,
        value TEXT NOT NULL
      );
    `);
  }

  insertSample(s: SampleRow): void {
    this.db
      .prepare(
        `INSERT INTO samples (ts, node, temp, humidity, rssi) VALUES (?, ?, ?, ?, ?)
         ON CONFLICT(ts, node) DO UPDATE SET
           temp = excluded.temp, humidity = excluded.humidity, rssi = excluded.rssi`
      )
      .run(s.ts, s.node, s.temp, s.humidity, s.rssi);
  }

  insertBoilerEvent(e: BoilerEventRow): void {
    this.db
      .prepare(
        `INSERT INTO boiler_events (ts, heat, pump, level, pump_level, reason)
         VALUES (?, ?, ?, ?, ?, ?)
         ON CONFLICT(ts) DO UPDATE SET
           heat = excluded.heat, pump = excluded.pump, level = excluded.level,
           pump_level = excluded.pump_level, reason = excluded.reason`
      )
      .run(e.ts, e.heat ? 1 : 0, e.pump ? 1 : 0, e.level, e.pumpLevel, e.reason);
  }

  series(from: number, to: number, node?: string): SampleRow[] {
    const sql = node
      ? `SELECT ts, node, temp, humidity, rssi FROM samples
         WHERE ts >= ? AND ts <= ? AND node = ? ORDER BY ts`
      : `SELECT ts, node, temp, humidity, rssi FROM samples
         WHERE ts >= ? AND ts <= ? ORDER BY ts`;
    const args = node ? [from, to, node] : [from, to];
    return this.db.prepare(sql).all(...args) as unknown as SampleRow[];
  }

  hourly(from: number, to: number): HourlyRow[] {
    const rows = this.db
      .prepare(
        `SELECT hour, node, temp_avg, temp_min, temp_max, humidity_avg
         FROM samples_hourly WHERE hour >= ? AND hour <= ? ORDER BY hour`
      )
      .all(from, to) as unknown as Record<string, number | string | null>[];
    return rows.map((r) => ({
      hour: r.hour as number,
      node: r.node as string,
      tempAvg: r.temp_avg as number,
      tempMin: r.temp_min as number,
      tempMax: r.temp_max as number,
      humidityAvg: r.humidity_avg as number | null,
    }));
  }

  /**
   * Минуты горения по календарным дням. Интервалы восстанавливаются из событий:
   * пара «heat=1 … heat=0». Незакрытый интервал считается до `to`.
   * Интервал, пересекающий полночь, режется по дням.
   */
  burnMinutesByDay(from: number, to: number): { day: string; minutes: number }[] {
    const rows = this.db
      .prepare(`SELECT ts, heat FROM boiler_events WHERE ts >= ? AND ts <= ? ORDER BY ts`)
      .all(from, to) as unknown as { ts: number; heat: number }[];

    const perDay = new Map<string, number>();
    const add = (start: number, end: number) => {
      let cursor = start;
      while (cursor < end) {
        const d = new Date(cursor);
        const midnight = new Date(d.getFullYear(), d.getMonth(), d.getDate() + 1, 0).getTime();
        const chunkEnd = Math.min(end, midnight);
        const day = localDay(cursor);
        perDay.set(day, (perDay.get(day) ?? 0) + (chunkEnd - cursor));
        cursor = chunkEnd;
      }
    };

    let onSince: number | null = null;
    for (const r of rows) {
      if (r.heat === 1 && onSince === null) onSince = r.ts;
      else if (r.heat === 0 && onSince !== null) {
        add(onSince, r.ts);
        onSince = null;
      }
    }
    if (onSince !== null) add(onSince, to);

    return [...perDay.entries()]
      .map(([day, ms]) => ({ day, minutes: Math.round(ms / 60000) }))
      .sort((a, b) => a.day.localeCompare(b.day));
  }

  /** Сырая запись в таблицу настроек — только для тестов порчи данных. */
  rawSet(key: string, value: string): void {
    this.db
      .prepare(`INSERT INTO settings (key, value) VALUES (?, ?)
                ON CONFLICT(key) DO UPDATE SET value = excluded.value`)
      .run(key, value);
  }

  loadSettings(rooms: string[]): Settings {
    const rows = this.db.prepare(`SELECT key, value FROM settings`).all() as unknown as {
      key: string;
      value: string;
    }[];
    const patch: Record<string, unknown> = {};
    for (const r of rows) {
      try {
        patch[r.key] = JSON.parse(r.value);
      } catch {
        // мусор в базе игнорируем — упадём на дефолт, но не уроним сервис
      }
    }
    try {
      return validateSettings(patch, rooms);
    } catch {
      // одно битое поле не должно стоить нам всех настроек: применяем по одному
      let acc: Settings = DEFAULT_SETTINGS;
      for (const [k, v] of Object.entries(patch)) {
        try {
          acc = validateSettings({ [k]: v }, rooms, acc);
        } catch {
          // поле пропускаем
        }
      }
      return acc;
    }
  }

  saveSettings(s: Settings): void {
    const stmt = this.db.prepare(
      `INSERT INTO settings (key, value) VALUES (?, ?)
       ON CONFLICT(key) DO UPDATE SET value = excluded.value`
    );
    for (const [k, v] of Object.entries(s)) stmt.run(k, JSON.stringify(v));
  }

  /** Сворачивает сырые замеры старше `keepRawDays` в часовые агрегаты. */
  compact(now: number, keepRawDays = 30): void {
    const cutoff = now - keepRawDays * 24 * 3_600_000;
    this.db
      .prepare(
        `INSERT INTO samples_hourly (hour, node, temp_avg, temp_min, temp_max, humidity_avg)
         SELECT (ts / 3600000) * 3600000 AS hour, node,
                AVG(temp), MIN(temp), MAX(temp), AVG(humidity)
         FROM samples WHERE ts < ?
         GROUP BY hour, node
         ON CONFLICT(hour, node) DO UPDATE SET
           temp_avg = excluded.temp_avg, temp_min = excluded.temp_min,
           temp_max = excluded.temp_max, humidity_avg = excluded.humidity_avg`
      )
      .run(cutoff);
    this.db.prepare(`DELETE FROM samples WHERE ts < ?`).run(cutoff);
  }

  close(): void {
    this.db.close();
  }
}
```

```ts
import { EventEmitter } from "node:events";
import mqtt, { MqttClient } from "mqtt";
import { Settings } from "@heating/shared";

export type Parsed =
  | { kind: "sensor"; node: string; metric: string }
  | { kind: "switch"; node: string; name: string }
  | { kind: "status"; node: string };

/** Служебные ветки, которые публикует сам сервис, — не данные нод. */
const OWN_BRANCHES = new Set(["cmd", "cfg"]);

export function parseTopic(prefix: string, topic: string): Parsed | null {
  const p = topic.split("/");
  if (p[0] !== prefix) return null;
  const node = p[1];
  if (!node || OWN_BRANCHES.has(node)) return null;

  if (p.length === 3 && p[2] === "status") return { kind: "status", node };
  if (p.length === 5 && p[4] === "state") {
    if (p[2] === "sensor") return { kind: "sensor", node, metric: p[3] };
    if (p[2] === "switch") return { kind: "switch", node, name: p[3] };
  }
  return null;
}

export interface HeatingMqttOptions {
  prefix: string;
  url?: string;
  username?: string;
  password?: string;
  /** подменяется в тестах */
  connect?: (url: string, opts: mqtt.IClientOptions) => MqttClient;
  now?: () => number;
}

export class HeatingMqtt extends EventEmitter {
  private client: MqttClient | null = null;
  private connected = false;
  private readonly now: () => number;

  constructor(private readonly opts: HeatingMqttOptions) {
    super();
    this.now = opts.now ?? Date.now;
  }

  get isConnected(): boolean {
    return this.connected;
  }

  start(): void {
    const connect = this.opts.connect ?? mqtt.connect;
    this.client = connect(this.opts.url ?? "", {
      username: this.opts.username,
      password: this.opts.password,
      reconnectPeriod: 5000,
    });

    this.client.on("connect", () => {
      this.connected = true;
      this.client?.subscribe(`${this.opts.prefix}/#`);
      this.emit("connected");
    });
    this.client.on("close", () => {
      this.connected = false;
      this.emit("disconnected");
    });
    this.client.on("error", (e: Error) => this.emit("mqtt-error", e));
    this.client.on("message", (topic: string, payload: Buffer) =>
      this.onMessage(topic, payload.toString())
    );
  }

  private onMessage(topic: string, raw: string): void {
    const parsed = parseTopic(this.opts.prefix, topic);
    if (!parsed) return;
    const at = this.now();

    if (parsed.kind === "status") {
      this.emit("status", parsed.node, raw.trim() === "online", at);
      return;
    }
    if (parsed.kind === "switch") {
      const on = raw.trim().toUpperCase() === "ON";
      this.emit("switch", parsed.node, parsed.name, on, at);
      return;
    }
    const value = Number(raw);
    if (!Number.isFinite(value)) return;
    this.emit("sensor", parsed.node, parsed.metric, value, at);
  }

  private pub(topic: string, payload: string, retain: boolean): void {
    if (!this.client || !this.connected) return;
    this.client.publish(topic, payload, { retain, qos: 0 });
  }

  /** Команда котлу. Без retain: свежесть сообщения — это и есть heartbeat сервиса. */
  publishHeat(on: boolean): void {
    this.pub(`${this.opts.prefix}/cmd/heat`, on ? "ON" : "OFF", false);
  }

  /** Конфигурация для автономной работы ноды. Всегда retained. */
  publishConfig(s: Settings): void {
    const p = this.opts.prefix;
    this.pub(`${p}/cfg/setpoint`, String(s.setpoint), true);
    this.pub(`${p}/cfg/hysteresis`, String(s.hysteresis), true);
    this.pub(`${p}/cfg/main_sensor`, s.mainSensor, true);
    this.pub(`${p}/cfg/pump_overrun`, String(s.pumpOverrunSeconds), true);
    this.pub(`${p}/cfg/pump_antiseize`, String(s.pumpAntiseizeSeconds), true);
  }

  stop(): void {
    this.client?.end();
    this.client = null;
    this.connected = false;
  }
}
```

```ts
export interface Alert {
  key: string;
  text: string;
}

export interface AlertContext {
  silentRooms: string[];
  boilerOfflineReported: boolean;
  pumpOfflineReported: boolean;

const NOT_FIRING_AFTER = 15 * MIN;
const NOT_FIRING_RISE = 5; // °C по подаче
const BURN_WATCH = 120 * MIN;
const BURN_MIN_RISE = 0.5; // °C по комнате
const WIDE_DELTA = 30; // °C между подачей и обраткой
const WIDE_DELTA_FOR = 10 * MIN;
const COLD_BELOW = 10;
const COLD_CLEAR = 11; // гистерезис, чтобы не звенеть на границе
const PUMP_GRACE = 2 * MIN;
const OVERRIDE_AFTER = 30 * MIN;

const LEVEL_TEXT: Record<string, string> = {
  L0: "Всё в норме: отоплением снова управляет сервис.",
  L1: "Сервис не отвечает — реле греет само по главному датчику.",
  L2: "Брокер недоступен — реле перешло на слепой цикл 15/45 минут.",

export function evaluateAlerts(
  snap: Snapshot,
  prev: AlertContext
): { alerts: Alert[]; ctx: AlertContext } {
  const alerts: Alert[] = [];
  const ctx: AlertContext = { ...prev, silentRooms: [...prev.silentRooms] };
  const now = snap.now;

  // 1. Молчащие датчики
  for (const room of snap.rooms) {
    const silent = !room.online || room.updatedAt === null || now - room.updatedAt > SILENT_AFTER;
    const known = ctx.silentRooms.includes(room.room);
    if (silent && !known) {
      ctx.silentRooms.push(room.room);
      alerts.push({
        key: `silent:${room.room}`,
        text: `Датчик «${room.title}» молчит больше 15 минут.`,
      });
    } else if (!silent && known) {
      ctx.silentRooms = ctx.silentRooms.filter((r) => r !== room.room);
      alerts.push({
        key: `silent-ok:${room.room}`,
        text: `Датчик «${room.title}» снова на связи.`,
      });
    }
  }

  // 2. Ноды
  if (!snap.boiler.online && !ctx.boilerOfflineReported) {
    ctx.boilerOfflineReported = true;
    alerts.push({ key: "boiler-offline", text: "Реле котла отвалилось от брокера." });
  } else if (snap.boiler.online && ctx.boilerOfflineReported) {
    ctx.boilerOfflineReported = false;
    alerts.push({ key: "boiler-online", text: "Реле котла снова на связи." });
  }

  if (!snap.pump.online && !ctx.pumpOfflineReported) {
    ctx.pumpOfflineReported = true;
    alerts.push({
      key: "pump-offline",
      text:
        "Нода насоса отвалилась от брокера, насос остановлен. " +
        "Тёплый пол остынет, батареи работают как обычно.",
    });
  } else if (snap.pump.online && ctx.pumpOfflineReported) {
    ctx.pumpOfflineReported = false;
    alerts.push({ key: "pump-online", text: "Нода насоса снова на связи." });
  }

  // 3. Уровни
  if (ctx.lastLevel !== null && snap.boiler.level !== ctx.lastLevel) {
    alerts.push({
      key: `level:${snap.boiler.level}`,
      text: LEVEL_TEXT[snap.boiler.level] ?? `Уровень ${snap.boiler.level}.`,
    });
  }
  ctx.lastLevel = snap.boiler.level;

  if (ctx.lastPumpLevel !== null && snap.pump.level !== ctx.lastPumpLevel) {
    alerts.push(
      snap.pump.level === "P1"
        ? {
            key: "pump-blind",
            text:
              "Нода насоса не видит состояния котла и остановила насос. Тёплый пол остынет.",
          }
        : {
            key: "pump-ok",
            text: "Нода насоса снова видит котёл, выбег работает штатно.",
          }
    );
  }
  ctx.lastPumpLevel = snap.pump.level;

  // 3b. Реле не слушается: команда сервиса расходится с фактом. Отдельного флага
  // «нажали кнопку» в MQTT нет — вычисляем сравнением (SPEC.md §5).
  if (snap.desiredHeat !== null && snap.desiredHeat !== snap.boiler.heat) {
    if (ctx.disagreeSince === null) ctx.disagreeSince = now;
    if (!ctx.overrideReported && now - ctx.disagreeSince >= OVERRIDE_AFTER) {
      ctx.overrideReported = true;
      alerts.push({
        key: "override-stuck",
        text: snap.boiler.heat
          ? "Котёл замкнут, хотя автоматика этого не просит. Похоже, нажали кнопку на коробке и забыли."
          : "Котёл разомкнут, хотя автоматика просит греть. Похоже, нажали кнопку на коробке и забыли.",
      });
    }
  } else {
    if (ctx.overrideReported) {
      alerts.push({ key: "override-gone", text: "Котёл снова слушается автоматики." });
    }
    ctx.disagreeSince = null;
    ctx.overrideReported = false;
  }

  // 4. Холодно в доме
  const main = snap.rooms.find((r) => r.room === snap.settings.mainSensor);
  const mainTemp = main?.temp ?? null;
  if (mainTemp !== null && mainTemp < COLD_BELOW && !ctx.coldReported) {
    ctx.coldReported = true;
    alerts.push({ key: "cold", text: `В доме ${mainTemp.toFixed(1)} °C — это ниже 10 °C.` });
  } else if (mainTemp !== null && mainTemp >= COLD_CLEAR) {
    ctx.coldReported = false;
  }

  // 5. Диагностика горения по подаче и обратке
  const flow = snap.boiler.flowTemp;
  const ret = snap.boiler.returnTemp;

  if (snap.boiler.heat) {
    if (ctx.burnStartedAt === null) {
      ctx.burnStartedAt = now;
      ctx.tempAtBurnStart = mainTemp;
      ctx.flowAtBurnStart = flow;
      ctx.notFiringReported = false;
      ctx.noProgressReported = false;
    } else {
      const burning = now - ctx.burnStartedAt;
      const flowRose =
        flow !== null && ctx.flowAtBurnStart !== null && flow - ctx.flowAtBurnStart >= NOT_FIRING_RISE;

      // 5a. Котёл вообще не запустился
      if (
        !ctx.notFiringReported &&
        burning >= NOT_FIRING_AFTER &&
        flow !== null &&
        ctx.flowAtBurnStart !== null &&
        !flowRose
      ) {
        ctx.notFiringReported = true;
        alerts.push({
          key: "not-firing",
          text:
            `Контакт замкнут пятнадцать минут, а подача не нагревается ` +
            `(${flow.toFixed(1)} °C). Котёл не запускается — проверь газ и ошибки на панели.`,
        });
      }

      // 5b. Котёл работает, а дом не греется
      if (
        !ctx.noProgressReported &&
        burning >= BURN_WATCH &&
        flowRose &&
        mainTemp !== null &&
        ctx.tempAtBurnStart !== null &&
        mainTemp - ctx.tempAtBurnStart < BURN_MIN_RISE
      ) {
        ctx.noProgressReported = true;
        alerts.push({
          key: "no-progress",
          text:
            `Котёл горит два часа и подача горячая, а в доме теплее не стало ` +
            `(${ctx.tempAtBurnStart.toFixed(1)} → ${mainTemp.toFixed(1)} °C). ` +
            `Проверь окна и воздух в системе.`,
        });
      }
    }

    // 5c. Дельта подача−обратка
    if (flow !== null && ret !== null && flow - ret > WIDE_DELTA) {
      if (ctx.wideDeltaSince === null) ctx.wideDeltaSince = now;
      if (!ctx.noCirculationReported && now - ctx.wideDeltaSince >= WIDE_DELTA_FOR) {
        ctx.noCirculationReported = true;
        alerts.push({
          key: "no-circulation",
          text:
            `Подача ${flow.toFixed(0)} °C, обратка ${ret.toFixed(0)} °C — дельта великовата. ` +
            `Похоже, циркуляция слабая или встала.`,
        });
      }
    } else {
      ctx.wideDeltaSince = null;
      ctx.noCirculationReported = false;
    }
  } else {
    ctx.burnStartedAt = null;
    ctx.tempAtBurnStart = null;
    ctx.flowAtBurnStart = null;
    ctx.notFiringReported = false;
    ctx.noProgressReported = false;
    ctx.wideDeltaSince = null;
    ctx.noCirculationReported = false;
  }

  // 6. Насос против котла
  if (snap.boiler.heat) {
    ctx.heatOffSince = null;
    ctx.pumpStuckReported = false;
    if (!snap.pump.on && snap.pump.online) {
      alerts.push({
        key: "pump-missing",
        text:
          "Котёл греет, а насос тёплого пола не включился — полы греться не будут. " +
          "На батареи это не влияет.",
      });
    }
  } else {
    if (ctx.heatOffSince === null) ctx.heatOffSince = now;
    const limit = snap.settings.pumpOverrunSeconds * 1000 + PUMP_GRACE;
    if (snap.pump.on && !ctx.pumpStuckReported && now - ctx.heatOffSince > limit) {
      ctx.pumpStuckReported = true;
      alerts.push({
        key: "pump-stuck",
        text: "Насос всё ещё крутится, хотя выбег давно закончился.",
      });
    }
    if (!snap.pump.on) ctx.pumpStuckReported = false;
  }

  return { alerts, ctx };
}
```

```ts
import { Alert } from "./alerts";

type FetchLike = (url: string, init: { method: string; headers: Record<string, string>; body: string }) => Promise<{ ok: boolean }>;

export interface NotifierOptions {
  token: string;
  chatId: string;
  /** сколько молчать по одному и тому же ключу */
  minIntervalMs: number;
  now?: () => number;
  fetchImpl?: FetchLike;
}

export class Notifier {
  private readonly sentAt = new Map<string, number>();
  private readonly now: () => number;
  private readonly doFetch: FetchLike;

  constructor(private readonly opts: NotifierOptions) {
    this.now = opts.now ?? Date.now;
    this.doFetch = opts.fetchImpl ?? (globalThis.fetch as unknown as FetchLike);
  }

  /** true — сообщение ушло; false — задавлено антиспамом или не доставлено. */
  async send(alert: Alert): Promise<boolean> {
    const last = this.sentAt.get(alert.key);
    const now = this.now();
    if (last !== undefined && now - last < this.opts.minIntervalMs) return false;

    try {
      const res = await this.doFetch(
        `https://api.telegram.org/bot${this.opts.token}/sendMessage`,
        {
          method: "POST",
          headers: { "content-type": "application/json" },
          body: JSON.stringify({
            chat_id: this.opts.chatId,
            text: alert.text,
            disable_notification: false,
          }),
        }
      );
      if (!res.ok) return false;
      this.sentAt.set(alert.key, now);
      return true;
    } catch {
      // Telegram недоступен — это не повод падать сервису отопления
      return false;
    }
  }
}
```

```ts
import { Store } from "./store";

const LOOPBACK = new Set(["127.0.0.1", "::1", "::ffff:127.0.0.1"]);

export function isLoopback(ip: string | undefined): boolean {
  return ip !== undefined && LOOPBACK.has(ip);
}

export interface ServerDeps {
  config: Config;
  store: Store;
  getSnapshot: () => Snapshot;
  applySettings: (s: Settings) => void;
}

export function createServer(deps: ServerDeps): {
  server: http.Server;
  broadcast: (s: Snapshot) => void;
} {
  const app = express();
  app.use(express.json());

  // Авторизации у нас нет — её делает Caddy через forward_auth и подкладывает
  // X-User/X-Role. Значит, обращения в обход прокси недопустимы: заголовкам можно
  // верить только если соединение пришло с петли. Fail closed.
  app.use((req, res, next) => {
    if (!isLoopback(req.socket.remoteAddress)) {
      return res.status(403).json({ ok: false, error: "Доступ только через reverse proxy" });
    }
    next();
  });

  const requireAdmin: RequestHandler = (req, res, next) => {
    if (req.header("x-role") !== "admin") {
      return res.status(403).json({ ok: false, error: "Требуется роль admin" });
    }
    next();
  };

  app.get("/heating/api/health", (_req, res) => res.json({ ok: true }));

  // Личность, как её видит Caddy. Отдельным маршрутом, а не полем снапшота:
  // снапшот один на всех и рассылается по WS, а роль у каждого своя.
  app.get("/heating/api/me", (req, res) =>
    res.json({ user: req.header("x-user") ?? "", role: req.header("x-role") ?? "viewer" })
  );

  app.get("/heating/api/state", (_req, res) => res.json(deps.getSnapshot()));
  app.get("/heating/api/settings", (_req, res) => res.json(deps.getSnapshot().settings));

  app.put("/heating/api/settings", requireAdmin, (req, res) => {
    try {
      const rooms = deps.config.rooms.map((r) => r.id);
      const next = validateSettings(req.body, rooms, deps.getSnapshot().settings);
      deps.applySettings(next);
      res.json(next);
    } catch (e) {
      res.status(400).json({ ok: false, error: (e as Error).message });
    }
  });

  app.get("/heating/api/history", (req, res) => {
    const from = Number(req.query.from);
    const to = Number(req.query.to);
    if (!Number.isFinite(from) || !Number.isFinite(to) || from > to) {
      return res.status(400).json({ ok: false, error: "Параметры from и to должны быть числами" });
    }
    const node = typeof req.query.node === "string" ? req.query.node : undefined;
    res.json({
      samples: deps.store.series(from, to, node),
      hourly: deps.store.hourly(from, to),
    });
  });

  app.get("/heating/api/stats/daily", (req, res) => {
    const from = Number(req.query.from);
    const to = Number(req.query.to);
    if (!Number.isFinite(from) || !Number.isFinite(to) || from > to) {
      return res.status(400).json({ ok: false, error: "Параметры from и to должны быть числами" });
    }
    res.json({ days: deps.store.burnMinutesByDay(from, to) });
  });

  // Статика Next: basePath '/heating' уже вшит в сборку, поэтому монтируем как есть.
  const publicDir = path.join(__dirname, "..", "..", "web", "out");
  app.use("/heating", express.static(publicDir, { extensions: ["html"] }));

  const server = http.createServer(app);
  const wss = new WebSocketServer({ server, path: "/heating/ws" });

  wss.on("connection", (ws, req) => {
    if (!isLoopback(req.socket.remoteAddress)) {
      ws.close(1008, "Доступ только через reverse proxy");
      return;
    }
    ws.send(JSON.stringify(deps.getSnapshot()));
  });

  const broadcast = (s: Snapshot) => {
    const payload = JSON.stringify(s);
    for (const ws of wss.clients) {
      if (ws.readyState === ws.OPEN) ws.send(payload);
    }
  };

  return { server, broadcast };
}
```

```ts
import path from "node:path";
import { Level, PumpLevel, RoomSample, Settings, Snapshot } from "@heating/shared";
import { AlertContext, EMPTY_ALERT_CONTEXT, evaluateAlerts } from "./alerts";
import { loadConfig } from "./config";
import { HeatingMqtt } from "./mqtt";
import { Notifier } from "./notify";
import { createServer } from "./server";
import { Store } from "./store";
import { decide } from "./thermostat";

const TICK_MS = 10_000;
const SAMPLE_MS = 60_000;
const COMPACT_MS = 6 * 3_600_000;
const TELEGRAM_QUIET_MS = 30 * 60_000;

const LEVELS: Level[] = ["L0", "L1", "L2", "L3"];
const PUMP_LEVELS: PumpLevel[] = ["P0", "P1"];

const config = loadConfig(process.env);
const store = new Store(path.join(config.dataDir, "heating.db"));
let settings: Settings = store.loadSettings(config.rooms.map((r) => r.id));

/** Живое состояние, собираемое из MQTT. */
const rooms = new Map<string, RoomSample>(
  config.rooms.map((r) => [
    r.id,
    { room: r.id, title: r.title, temp: null, humidity: null, rssi: null, online: false, updatedAt: null },
  ])
);

const boiler = {
  online: false,
  heat: false,
  level: "L0" as Level,
  boilerRoomTemp: null as number | null,
  outdoorTemp: null as number | null,
  flowTemp: null as number | null,
  returnTemp: null as number | null,
  updatedAt: null as number | null,
};

const pump = {
  online: false,
  on: false,
  overrunLeft: 0,
  level: "P0" as PumpLevel,
  updatedAt: null as number | null,
};

let lastHeatChangeAt: number | null = null;
let lastReason = "запуск";
let lastDesired: boolean | null = null;
let alertCtx: AlertContext = EMPTY_ALERT_CONTEXT;

const mqttClient = new HeatingMqtt({
  prefix: config.mqttTopicPrefix,
  url: config.mqttUrl,
  username: config.mqttUser,
  password: config.mqttPassword,
});

const notifier = config.telegram
  ? new Notifier({ ...config.telegram, minIntervalMs: TELEGRAM_QUIET_MS })
  : null;

mqttClient.on("connected", () => {
  console.log("MQTT подключён");
  mqttClient.publishConfig(settings);
});
mqttClient.on("mqtt-error", (e: Error) => console.error("MQTT:", e.message));

function writeEvent(at: number): void {
  store.insertBoilerEvent({
    ts: at,
    heat: boiler.heat,
    pump: pump.on,
    level: boiler.level,
    pumpLevel: pump.level,
    reason: lastReason,
  });
}

mqttClient.on("sensor", (node: string, metric: string, value: number, at: number) => {
  if (node === "boiler") {
    if (metric === "boilerroom_temperature") boiler.boilerRoomTemp = value;
    else if (metric === "outdoor_temperature") boiler.outdoorTemp = value;
    else if (metric === "flow_temperature") boiler.flowTemp = value;
    else if (metric === "return_temperature") boiler.returnTemp = value;
    // Уровень едет числом 0..3, а не строкой «L0»: числовой канал уже разбирается
    // общим парсером, отдельный текстовый ради одного значения не нужен.
    else if (metric === "level") boiler.level = LEVELS[value] ?? "L0";
    // Нода котла шлёт heat числом каждые 10 с — это heartbeat для ноды насоса.
    // Нам оно ничего не даёт сверх switch/heat/state, поэтому просто игнорируем.
    boiler.updatedAt = at;
    return;
  }
  if (node === "pump") {
    if (metric === "overrun_left") pump.overrunLeft = value;
    else if (metric === "level") pump.level = PUMP_LEVELS[value] ?? "P0";
    pump.updatedAt = at;
    return;
  }
  const room = rooms.get(node);
  if (!room) return;
  if (metric === "temperature") {
    room.temp = value;
    room.updatedAt = at;
  } else if (metric === "humidity") room.humidity = value;
  else if (metric === "wifi_signal") room.rssi = value;
});

mqttClient.on("switch", (node: string, name: string, on: boolean, at: number) => {
  if (node === "boiler" && name === "heat") {
    if (boiler.heat !== on) lastHeatChangeAt = at;
    boiler.heat = on;
    boiler.updatedAt = at;
    writeEvent(at);
  } else if (node === "pump" && name === "pump") {
    pump.on = on;
    pump.updatedAt = at;
    writeEvent(at);
  }
});

mqttClient.on("status", (node: string, online: boolean) => {
  if (node === "boiler") boiler.online = online;
  else if (node === "pump") pump.online = online;
  else {
    const room = rooms.get(node);
    if (room) room.online = online;
  }
});

function snapshot(now: number): Snapshot {
  const main = rooms.get(settings.mainSensor) ?? null;
  const d = decide(
    {
      now,
      mainTemp: main?.temp ?? null,
      mainTempAt: main?.updatedAt ?? null,
      currentHeat: boiler.heat,
      lastChangeAt: lastHeatChangeAt,
    },
    settings
  );
  lastReason = d.reason;
  lastDesired = d.command === "none" ? null : d.command === "on";
  return {
    now,
    rooms: [...rooms.values()],
    boiler: { ...boiler },
    pump: { ...pump },
    settings,
    desiredHeat: lastDesired,
    reason: d.reason,
  };
}

const { server, broadcast } = createServer({
  config,
  store,
  getSnapshot: () => snapshot(Date.now()),
  applySettings: (s) => {
    settings = s;
    store.saveSettings(s);
    mqttClient.publishConfig(s);
  },
});

mqttClient.start();
server.listen(config.port, config.bind, () => {
  console.log(`heating-service слушает http://${config.bind}:${config.port}/heating`);
});

// Основной цикл: решение, команда, тревоги, рассылка снапшота.
setInterval(() => {
  const snap = snapshot(Date.now());
  if (snap.desiredHeat !== null) mqttClient.publishHeat(snap.desiredHeat);
  const { alerts, ctx } = evaluateAlerts(snap, alertCtx);
  alertCtx = ctx;
  if (notifier && settings.telegramEnabled) {
    for (const a of alerts) void notifier.send(a);
  }
  broadcast(snap);
}, TICK_MS);

// Запись истории. Котельная и улица идут как отдельные «ноды» в samples.
setInterval(() => {
  const now = Date.now();
  for (const r of rooms.values()) {
    if (r.temp === null) continue;
    store.insertSample({ ts: now, node: r.room, temp: r.temp, humidity: r.humidity, rssi: r.rssi });
  }
  const extra: [string, number | null][] = [
    ["boilerroom", boiler.boilerRoomTemp],
    ["outdoor", boiler.outdoorTemp],
    ["flow", boiler.flowTemp],
    ["return", boiler.returnTemp],
  ];
  for (const [node, temp] of extra) {
    if (temp === null) continue;
    store.insertSample({ ts: now, node, temp, humidity: null, rssi: null });
  }
}, SAMPLE_MS);

// Сворачивание старых данных.
setInterval(() => store.compact(Date.now()), COMPACT_MS);

for (const sig of ["SIGINT", "SIGTERM"] as const) {
  process.on(sig, () => {
    console.log(`Получен ${sig}, останавливаюсь`);
    mqttClient.stop();
    server.close();
    store.close();
    process.exit(0);
  });
}
```

```tsx
"use client";

import { useEffect, useRef } from "react";
import uPlot from "uplot";
import "uplot/dist/uPlot.min.css";

export interface Series {
  label: string;
  color: string;
  /** [секунды unix, значение] */
  points: [number, number][];
}

export function TempChart({ series, height = 220 }: { series: Series[]; height?: number }) {
  const box = useRef<HTMLDivElement>(null);

  useEffect(() => {
    if (!box.current || series.length === 0) return;

    const xs = [...new Set(series.flatMap((s) => s.points.map((p) => p[0])))].sort((a, b) => a - b);
    const index = new Map(xs.map((x, i) => [x, i]));
    const data: uPlot.AlignedData = [
      xs,
      ...series.map((s) => {
        const col = new Array<number | null>(xs.length).fill(null);
        for (const [x, y] of s.points) col[index.get(x)!] = y;
        return col;
      }),
    ];

    const chart = new uPlot(
      {
        width: box.current.clientWidth,
        height,
        series: [{}, ...series.map((s) => ({ label: s.label, stroke: s.color, width: 2 }))],
        axes: [{}, { label: "°C" }],
      },
      data,
      box.current
    );
    const onResize = () => chart.setSize({ width: box.current!.clientWidth, height });
    window.addEventListener("resize", onResize);
    return () => {
      window.removeEventListener("resize", onResize);
      chart.destroy();
    };
  }, [series, height]);

  return <div ref={box} />;
}
```

```ts
async function json<T>(res: Response): Promise<T> {
  if (!res.ok) {
    const body = (await res.json().catch(() => ({}))) as { error?: string };
    throw new Error(body.error ?? `Ошибка ${res.status}`);
  }
  return (await res.json()) as T;
}

  role: string;
}

  state: () => fetch(`${BASE}/state`).then(json<Snapshot>),
```

```ts
"use client";

import { useEffect, useState } from "react";
import { Snapshot } from "@heating/shared";
import { api } from "./api";

/** Живой снапшот: WS, а если он не поднялся — опрос раз в 10 секунд. */
export function useSnapshot(): { snap: Snapshot | null; error: string | null } {
  const [snap, setSnap] = useState<Snapshot | null>(null);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    let poll: ReturnType<typeof setInterval> | null = null;
    api.state().then(setSnap).catch((e: Error) => setError(e.message));

    const proto = window.location.protocol === "https:" ? "wss" : "ws";
    const ws = new WebSocket(`${proto}://${window.location.host}/heating/ws`);
    ws.onmessage = (ev) => setSnap(JSON.parse(ev.data as string) as Snapshot);
    ws.onerror = () => {
      if (poll === null) {
        poll = setInterval(() => {
          api.state().then(setSnap).catch(() => undefined);
        }, 10_000);
      }
    };

    return () => {
      ws.close();
      if (poll !== null) clearInterval(poll);
    };
  }, []);

  return { snap, error };
}

/** Роль текущего пользователя. Пока не загрузилась — считаем viewer: fail closed. */
export function useRole(): string {
  const [role, setRole] = useState("viewer");
  useEffect(() => {
    api
      .me()
      .then((m) => setRole(m.role))
      .catch(() => undefined);
  }, []);
  return role;
}
```

```ts
// Точка для forward_auth в Caddy: отдаёт личность заголовками, а не телом,
// потому что copy_headers умеет копировать только заголовки. Смонтирована до
// authenticate намеренно — ей нужна своя логика отказа.
app.get("/api/authz", (req, res) => {
  const s = auth.verify(reqToken(req));
  const t = s ? null : auth.verifyToken(bearerFromHeader(req.headers.authorization));
  const who = s ?? t;
  if (!who) {
    if (req.query.on_fail === "redirect") return res.redirect("/login");
    return res.status(401).json({ ok: false, error: "Unauthorized" });
  }
  res.setHeader("X-User", who.username);
  res.setHeader("X-Role", who.role);
  res.status(204).end();
});
```

```bash
# Деплой heating на Raspberry Pi.

echo "==> Установка и рестарт на Pi"
"${SSH[@]}" "$PI_HOST" bash -s <<EOF
set -euo pipefail
cd "$PI_DIR"
npm ci -w server --omit=dev
sudo cp infra/heating.service /etc/systemd/system/heating.service
sudo systemctl daemon-reload
```

## Задача 16: Прошивка ноды котла

**Файлы:**

**Интерфейсы:**
- Публикует: `heating/boiler/status`, `heating/boiler/switch/heat/state`,
  `heating/boiler/sensor/heat/state` (числом, каждые 10 с),
  `heating/boiler/sensor/{boilerroom,outdoor,flow,return}_temperature/state`,
  `heating/boiler/sensor/level/state`
- Подписан: `heating/cmd/heat`, `heating/cfg/{setpoint,hysteresis,main_sensor}`,

Два момента, которых не было в прежней версии. Реле здесь **одно** — насосом
занимается отдельная нода. И состояние котла публикуется **числом каждый тик**, а
не только при изменении: на свежести этого топика держится вся логика ноды насоса

улица отдельно на `GPIO17`. Разделение намеренное — обрыв уличного кабеля не должен
уносить датчик котельной, от которого зависит уровень L3.

```yaml
  - platform: mqtt_subscribe
    id: sub_t2
    name: "детская (вход)"
    internal: true
    topic: heating/kids/sensor/temperature/state
    on_value:
      then:
        - lambda: |-
            if (id(cfg_main_idx) == 2) { id(room_temp) = x; id(room_temp_at) = millis(); }

text_sensor:
  - platform: mqtt_subscribe
    id: sub_cmd
    name: "команда (вход)"
    internal: true
    topic: heating/cmd/heat
    on_value:
      then:
        - lambda: |-
            id(last_cmd_on) = (x == "ON");
            id(last_cmd_at) = millis();

  - platform: mqtt_subscribe
    id: sub_main
    name: "главный датчик (вход)"
    internal: true
    topic: heating/cfg/main_sensor
    on_value:
      then:
        - lambda: |-
            if (x == "livingroom") id(cfg_main_idx) = 0;
            else if (x == "bedroom") id(cfg_main_idx) = 1;
            else if (x == "kids") id(cfg_main_idx) = 2;
            // смена главного датчика обесценивает кэш: до первого нового значения
            // считаем, что комнатной температуры у нас нет
            id(room_temp) = NAN;
            id(room_temp_at) = 0;

interval:
  - interval: 10s
    then:
      - lambda: |-
          const uint32_t now = millis();
          const uint32_t CMD_TTL   = 300000UL;    // 5 мин — порог L1
          const uint32_t TEMP_TTL  = 1800000UL;   // 30 мин — порог L2
          const uint32_t MIN_STATE = 300000UL;    // 5 мин — защита горелки
          const uint32_t L2_ON     = 15UL * 60000UL;
          const uint32_t L2_PERIOD = 60UL * 60000UL;

          const bool cmd_fresh  = id(last_cmd_at)  != 0 && (now - id(last_cmd_at))  < CMD_TTL;
          const bool temp_fresh = id(room_temp_at) != 0 && (now - id(room_temp_at)) < TEMP_TTL
                                  && !isnan(id(room_temp));

          bool heat;
          int level;

          if (cmd_fresh) {
            level = 0;
            heat = id(last_cmd_on);
            id(l2_since) = 0;
          } else if (temp_fresh) {
            level = 1;
            id(l2_since) = 0;
            const float half = id(cfg_hysteresis) / 2.0f;
            if (id(room_temp) <= id(cfg_setpoint) - half)      heat = true;
            else if (id(room_temp) >= id(cfg_setpoint) + half) heat = false;
            else                                               heat = id(relay_heat).state;
          } else {
            level = 2;
            if (id(l2_since) == 0) id(l2_since) = now;
            heat = ((now - id(l2_since)) % L2_PERIOD) < L2_ON;
          }

          // Защита горелки: не переключаться чаще, чем раз в 5 минут.
          // Действует и на кнопку тоже.
          if (heat != id(relay_heat).state && id(heat_changed_at) != 0
              && (now - id(heat_changed_at)) < MIN_STATE) {
            heat = id(relay_heat).state;
          }

          // Кнопка: переключённое ею состояние держится, пока автомат не поменяет
          // своё решение (SPEC.md §6). Сравниваем с тем, что автомат решал в прошлый
          // тик, — как только оно изменилось, флаг снимается.
          const int auto_now = heat ? 1 : 0;
          if (id(auto_prev) != -1 && auto_now != id(auto_prev)) id(override_on) = false;
          id(auto_prev) = auto_now;
          if (id(override_on)) heat = id(override_state);

          // L3 — жёсткая страховка поверх всего, включая защиту горелки.
          // Пороги зашиты намеренно: их нельзя сломать из веба.
          const float br = id(boilerroom_temp).state;
          if (!isnan(br)) {
            if (br < 8.0f)       { heat = true;  level = 3; }
            else if (br > 45.0f) { heat = false; level = 3; }
          }

          if (heat != id(relay_heat).state) {
            id(heat_changed_at) = now;
            if (heat) id(relay_heat).turn_on(); else id(relay_heat).turn_off();
          }

          id(level_num).publish_state(level);
          // Каждый тик, безусловно — это heartbeat для ноды насоса.
          id(heat_num).publish_state(heat ? 1 : 0);

          if (heat) id(led).turn_on(); else id(led).turn_off();
```

## Задача 17: Прошивка ноды насоса

**Файлы:**
- Создать: `firmware/pump-node.yaml`

**Интерфейсы:**
- Публикует: `heating/pump/status`, `heating/pump/switch/pump/state`,
  `heating/pump/sensor/overrun_left/state`, `heating/pump/sensor/level/state`
- Подписан: `heating/boiler/sensor/heat/state`, `heating/cfg/pump_overrun`,
  `heating/cfg/pump_antiseize`

Нода простая: одно реле, ни одного датчика, автомат из двух уровней. И она **не
подписана ни на одну команду сервиса** — зависит только от ноды котла и от
конфигурации.

Ключевое поведение: потеряв состояние котла дольше чем на минуту, нода **выключает**
насос. Обоснование в §6 спецификации — насос обслуживает только тёплый пол, крутить
его вслепую бессмысленно.

- [ ] **Шаг 1: Написать `firmware/pump-node.yaml`**

```yaml

  friendly_name: "Насос тёплого пола"

  framework:

  password: !secret mqtt_password_pump

  reboot_timeout: 0s

    retain: true

    type: int
    restore_value: yes
    initial_value: '900'
  - id: cfg_antiseize
    type: int
    restore_value: yes
    initial_value: '604800'

    type: bool
    restore_value: no
    initial_value: 'false'
  - id: boiler_heat_at
    type: uint32_t
    restore_value: no
    initial_value: '0'

    type: uint32_t
    restore_value: no
    initial_value: '0'
  - id: pump_last_on_at
    type: uint32_t
    restore_value: no
    initial_value: '0'

  - id: override_on
    type: bool
    restore_value: no
    initial_value: 'false'
  - id: override_state
    type: bool
    restore_value: no
    initial_value: 'false'

  - id: auto_prev
    type: int
    restore_value: no

  - platform: gpio
    id: relay_pump
    name: "насос"
    pin:
      number: GPIO16
      inverted: true
    restore_mode: ALWAYS_OFF
    state_topic: heating/pump/switch/pump/state

binary_sensor:
  - platform: gpio
    id: btn
    name: "кнопка"
    internal: true
    pin:
      number: GPIO27
      mode:
        input: true
        pullup: true
      inverted: true
    filters:

    on_press:

    update_interval: never

    update_interval: never
    state_topic: heating/pump/sensor/overrun_left/state

    name: "состояние котла (вход)"
    internal: true

    on_value:
      then:

            id(boiler_heat) = (x > 0.5f);
            id(boiler_heat_at) = millis();

    id: sub_overrun
    name: "выбег (вход)"
    internal: true
    topic: heating/cfg/pump_overrun

      then:
        - lambda: 'id(cfg_overrun) = (int) x;'

    id: sub_antiseize

    on_value:
      then:
        - lambda: 'id(cfg_antiseize) = (int) x;'

interval:
  - interval: 10s
    then:
      - lambda: |-
          const uint32_t now = millis();
          const uint32_t HEAT_TTL = 60000UL;   // минута — порог P1

          const bool fresh = id(boiler_heat_at) != 0
                             && (now - id(boiler_heat_at)) < HEAT_TTL;

          bool on;
          int level;

          if (fresh) {
            level = 0;
            if (id(boiler_heat)) {
              id(pump_until) = now + (uint32_t) id(cfg_overrun) * 1000UL;
            }
            on = id(boiler_heat)
                 || (id(pump_until) != 0 && (int32_t)(id(pump_until) - now) > 0);

            // Летний прогон: не даём насосу закиснуть. Только на уровне P0 —
            // в P1 нам всё равно нечем принимать решение.
            if (!on && id(cfg_antiseize) > 0
                && (now - id(pump_last_on_at)) > (uint32_t) id(cfg_antiseize) * 1000UL) {
              id(pump_until) = now + 180000UL;   // 3 минуты
              on = true;
            }
          } else {
            // P1: котла не слышно. Выключаемся — крутить вслепую бессмысленно,
            // тепла в контуре без горящего котла всё равно нет (SPEC.md §6).
            level = 1;
            on = false;
            id(pump_until) = 0;
          }

          // Кнопка: держится, пока автомат не поменяет решение (SPEC.md §6).
          const int auto_now = on ? 1 : 0;
          if (id(auto_prev) != -1 && auto_now != id(auto_prev)) id(override_on) = false;
          id(auto_prev) = auto_now;
          if (id(override_on)) on = id(override_state);

          if (on) id(pump_last_on_at) = now;

          if (on != id(relay_pump).state) {
            if (on) id(relay_pump).turn_on(); else id(relay_pump).turn_off();
          }

          id(level_num).publish_state(level);
          const int32_t left = (int32_t)(id(pump_until) - now) / 1000;
          id(overrun_left).publish_state(
            (on && !id(boiler_heat) && left > 0) ? left : 0);

          if (on) id(led).turn_on(); else id(led).turn_off();
```

## Задача 18: Стендовая проверка обеих лестниц

Отдельная задача, а не шаг предыдущей: это единственное место, где вообще можно
проверить код, который сработает один раз за зиму. На живом котле его не отладить.

Собери стенд: **обе ноды рядом на столе**, брокер на Pi, сервис запущен, все три
датчика включены. Реле без нагрузки.

Для скорости на время проверок уменьши пороги в lambda обеих нод (`CMD_TTL` до 30 с,
`TEMP_TTL` до 60 с, `MIN_STATE` до 20 с, `L2_ON`/`L2_PERIOD` до 30 с / 60 с,
`HEAT_TTL` до 15 с, выбег до 60 с), прошей, проверь — и **верни обратно**, прошив
заново. Последняя проверка каждого сценария делается уже на боевых порогах.

### Лестница котла

- [ ] **Шаг 1: L0 — обычная работа**

Опусти уставку ниже комнатной, потом подними выше.
Ожидаемо: реле щёлкает, `heating/boiler/sensor/level/state` = `0`, в вебе «горит».

- [ ] **Шаг 2: L1 — сервис умер**

```bash
ssh pi@192.168.1.112 'sudo systemctl stop heating'
```

Ожидаемо: через 5 минут `level` = `1`, реле продолжает поддерживать уставку.
Проверь обе стороны: подышав на датчик главной комнаты, добейся включения и
выключения **без работающего сервиса**.

```bash
ssh pi@192.168.1.112 'sudo systemctl start heating'
```

Ожидаемо: `level` возвращается в `0` в течение 30 секунд.

- [ ] **Шаг 3: L2 — брокер умер**

```bash
ssh pi@192.168.1.112 'sudo systemctl stop mosquitto'
```

Ожидаемо: через 30 минут `level` уходит в `2` (увидеть это можно только в логе ноды

```bash
ssh pi@192.168.1.112 'sudo systemctl start mosquitto'
```

- [ ] **Шаг 4: L3 — страховка**

Зажми `DS18B20` котельной в кулаке или окуни в тёплую воду выше 45 °C.

Затем охлади ниже 8 °C (морозилка, стакан со льдом) — реле замыкается.

- [ ] **Шаг 5: Heartbeat состояния котла**

Ожидаемо: строка **каждые 10 секунд** при неподвижном котле, а не только при

### Лестница насоса

- [ ] **Шаг 6: P0 — насос следует за котлом**

Включи котёл. Ожидаемо: реле насоса щёлкает в течение десяти секунд,

- [ ] **Шаг 8: Сброс таймера выбега**

Во время выбега снова включи котёл. Ожидаемо: таймер сбрасывается на полные

- [ ] **Шаг 9: P1 — главная проверка после разделения нод**


Ожидаемо: через минуту нода насоса размыкает реле и больше его не трогает,

```bash
ssh pi@192.168.1.112 'sudo systemctl start mosquitto'
```

Ожидаемо: возврат в `P0` и поведение по состоянию котла.

- [ ] **Шаг 10: Залипший retained не обманывает насос**

Замкни котёл, дождись `heat = 1`, затем **выключи ноду котла по питанию**, оставив

Ожидаемо: нода насоса через минуту всё равно уходит в P1 и выключается — она смотрит
на свежесть, а не только на значение. Это единственная проверка, которая ловит ошибку
«забыли учесть время последнего сообщения».

состоянию котла. Безусловного прогона на старте быть **не должно** — его убрали
намеренно (§6 спецификации).

- [ ] **Шаг 12: Летний прогон**

Временно выставь `heating/cfg/pump_antiseize` в 120 секунд, оставь котёл выключенным.
Ожидаемо: через 2 минуты простоя насос включается на 3 минуты. Верни `604800`.

### Кнопка

- [ ] **Шаг 12а: Нажатие переключает и удерживает**

Нажми кнопку на коробке котла при работающем автомате.
Ожидаемо: реле переключается **и остаётся** переключённым, а не возвращается через

Это главная проверка: если реле щёлкает обратно на следующем тике, значит флаг
`override_on` не работает и кнопка бесполезна.

- [ ] **Шаг 12б: Автомат забирает управление, когда передумает**

снимается и дальше рулит он.

- [ ] **Шаг 12в: Повторное нажатие переключает обратно**

Ожидаемо: реле возвращается, флаг снова поднят. Никаких долгих нажатий не нужно.

- [ ] **Шаг 12г: L3 выше кнопки**

Кнопкой **замкни** котёл, затем нагрей `DS18B20` котельной выше 45 °C.
Ожидаемо: реле размыкается, несмотря на нажатую кнопку. Единственное исключение из
правила «кнопка держится».

- [ ] **Шаг 12д: Флаг не переживает перезагрузку**

Нажми кнопку, затем передёрни питание ноды. Ожидаемо: после загрузки состояние
определяет автомат.

- [ ] **Шаг 12е: То же самое на ноде насоса**

Нажатие, удержание состояния, снятие флага при смене решения автомата. У насоса L3
нет, так что кнопка там ничем не перебивается.

- [ ] **Шаг 12ж: Сервис замечает забытую кнопку**

Кнопкой замкни котёл, когда автомат просит «не грей», и оставь так. Ожидаемо: через
30 минут в Telegram приходит `override-stuck`. Отдельного флага в MQTT нет — сервис
вычисляет расхождение сам, сравнивая свою команду с состоянием реле.

```tsx
"use client";

const MIN = 5;
const MAX = 30;
const STEP = 0.5;

export function SetpointControl({
  value,
  onChange,
  disabled,
}: {
  value: number;
  onChange: (next: number) => void;
  disabled: boolean;
}) {
  const bump = (delta: number) => {
    const next = Math.round((value + delta) * 2) / 2;
    if (next < MIN || next > MAX) return;
    onChange(next);
  };

  const btn = {
    fontSize: 28,
    width: 56,
    height: 56,
    borderRadius: 8,
    border: "1px solid #ccc",
    background: "#fff",
  } as const;

  return (
    <div style={{ display: "flex", alignItems: "center", gap: 12 }}>
      <button aria-label="холоднее" style={btn} disabled={disabled} onClick={() => bump(-STEP)}>
        −
      </button>
      <span style={{ fontSize: 24, minWidth: 100, textAlign: "center" }}>
        {value.toFixed(1)} °C
      </span>
      <button aria-label="теплее" style={btn} disabled={disabled} onClick={() => bump(STEP)}>
        +
      </button>
    </div>
  );
}
```

```tsx
"use client";

import { useEffect, useState } from "react";
import { AppHeader } from "@/components/AppHeader";
import { Series, TempChart } from "@/components/TempChart";
import { api } from "@/lib/api";
import { useSnapshot } from "@/lib/useSnapshot";

const COLORS = ["#1c7c3f", "#1f6fb2", "#b06a00", "#8e44ad"];
const RANGES = [
  { label: "Сутки", hours: 24 },
  { label: "Неделя", hours: 24 * 7 },
  { label: "Месяц", hours: 24 * 30 },
];

export default function HistoryPage() {
  const { snap } = useSnapshot();
  const [hours, setHours] = useState(24);
  const [series, setSeries] = useState<Series[]>([]);
  const [days, setDays] = useState<{ day: string; minutes: number }[]>([]);

  useEffect(() => {
    if (!snap) return;
    const to = Date.now();
    const from = to - hours * 3_600_000;

    api.history(from, to).then((res) => {
      const byNode = new Map<string, [number, number][]>();
      for (const s of res.samples) {
        if (s.temp === null) continue;
        const list = byNode.get(s.node) ?? [];
        list.push([Math.floor(s.ts / 1000), s.temp]);
        byNode.set(s.node, list);
      }
      const titles = new Map(snap.rooms.map((r) => [r.room, r.title]));
      setSeries(
        [...byNode.entries()].map(([node, points], i) => ({
          label: titles.get(node) ?? node,
          color: COLORS[i % COLORS.length],
          points,
        }))
      );
    });

    fetch(`/heating/api/stats/daily?from=${from}&to=${to}`)
      .then((r) => r.json())
      .then((r: { days: { day: string; minutes: number }[] }) => setDays(r.days));
  }, [snap, hours]);

  return (
    <>
      <AppHeader active="heating" />
      <h1>История</h1>
      <div style={{ display: "flex", gap: 8, marginBottom: 12 }}>
        {RANGES.map((r) => (
          <button
            key={r.hours}
            onClick={() => setHours(r.hours)}
            style={{ fontWeight: hours === r.hours ? 700 : 400, padding: "6px 12px" }}
          >
            {r.label}
          </button>
        ))}
      </div>

      <TempChart series={series} />

      <h2 style={{ marginTop: 24 }}>Часы горения по дням</h2>
      <table>
        <tbody>
          {days.map((d) => (
            <tr key={d.day}>
              <td style={{ paddingRight: 12 }}>{d.day}</td>
              <td style={{ paddingRight: 12 }}>{(d.minutes / 60).toFixed(1)} ч</td>
              <td>
                <div
                  style={{
                    width: Math.round((d.minutes / (24 * 60)) * 300),
                    height: 12,
                    background: "#1c7c3f",
                  }}
                />
              </td>
            </tr>
          ))}
        </tbody>
      </table>
    </>
  );
}
```
